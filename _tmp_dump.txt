000: // Source node (simple: downstream READY -> supply immediately)
001: /*
002:  * SourceNode（供給ノード）
003:  * - 下流が IDLE/READY のときだけ、即座に1件供給して ID を +1 する。
004:  * - インターバルや pending バッファは使用しない、最小仕様。
005:  */
006: 
007: class SourceNode extends LiteGraph.LGraphNode{                 // Source ノードの定義（ワーク供給専用）
008:   constructor(){
009:     super();                                                   // 親クラス(LiteGraph.LGraphNode)の初期化
010:     this.title = 'Source';                                     // ノードタイトル
011:     this.size = [240,180];                                     // 表示サイズ [幅, 高さ]
012:     this.addOutput('workOut', 0);                              // 出力0: ワークを流すポート
013:     // Fixed color: orange (Source has no state machine)
014:     this.color = '#f39c12';   // border (orange)
015:     this.bgcolor = '#fff6e6'; // fill   (light orange)
016:     this.properties = {                                        // ノード設定（プロパティ）
017:       sequence: 'A,B',                                         // 供給するタイプの並び（カンマ/改行区切り）
018:     };
019:     this._seq = [];                                            // 解析済み sequence の配列（{type}）
020:     this._cursor = 0;                                          // 次に供給する type のインデックス
021:     this._readyPrev = false;                                   // 前回 onExecute 時の下流 READY 状態（現仕様では未使用）
022:     this._parseSeq();                                          // 初期の sequence を解析
023:   }
024:   _parseSeq(){
025:     this._seq = [];                                            // 配列を初期化
026:     this._cursor = 0;                                          // 先頭から供給を開始
027:     // sequence 文字列をカンマ/改行で分割し、空白を除去して {type} として格納
028:     this.properties.sequence.split(/[\,\n]+/).forEach(t=>{ t=t.trim(); if(t) this._seq.push({type:t}); });
029:     if(!this._seq.length) this._seq.push({type:'A'});          // 空ならデフォルトで 'A' を1件入れる
030:   }
031:   onPropertyChanged(n){                                        // プロパティ変更時のハンドラ
032:     if(n==='sequence') this._parseSeq();                       // sequence 変更 → 再解析
033:   }
034:   onExecute(){
035:     // 下流が READY/IDLE かを判定（_state を持たないノード = 常時READY とみなす）
036:     let ready = false;                 // 受け渡し可能かどうかの最終判定フラグ
037:     let statelessReady = false;        // Sink 等（_state 未定義）で READY の場合に true
038:     const out = this.outputs && this.outputs[0]; // 出力0ポートの参照（存在しない場合は false）
039:     if(out && out.links){              // 出力ポートがあり、少なくとも1本リンクされている
040:       for(const lid of out.links){     // すべてのリンクIDを走査
041:         const link = this.graph.links[lid]; if(!link) continue;          // リンク実体を取得（無ければスキップ）
042:         const t = this.graph.getNodeById(link.target_id); if(!t) continue; // 接続先ノードを取得（無ければスキップ）
043:         if(typeof t._state === 'undefined'){          // _state を持たない（= stateless。例: Sink）
044:           ready = true;                                // 常時 READY とみなす
045:           statelessReady = true;                      // stateless の READY であることを記録
046:           break;                                      // 判定終了
047:         }
048:         if(t._state === 'IDLE'){                     // stateful ノード（例: Equipment）が IDLE
049:           ready = true;                               // 受け渡し可能
050:           /* 継続探索しない */ break;                // ひとつ READY が見つかれば十分
051:         }
052:       }
053:     }
054: 
055:     // 供給条件: 下流が READY/IDLE であれば即座に供給（Equipment は供給直後に PROCESS へ遷移し、
056:     // 次フレームには READY でなくなるため、多重供給は起きません）
057:     if(ready){                                 // 下流が READY/IDLE なら供給する
058:       if(!this._seq || !this._seq.length)      // まだ sequence が用意されていなければ
059:         this._parseSeq();                      // sequence を解析して初期化
060:       const e = this._seq[this._cursor] || {type:'A'};      // 現在カーソルの type（なければ 'A'）
061:       const w = new Work(workCounter , e.type);          // 供給直前に ID を付与して Work を生成
062:       this.setOutputData(0, w);                              // 出力0（workOut）に供給
063:       workCounter += 1;                                      // Work ID を1つ進める
064:       this._cursor = (this._cursor + 1) % this._seq.length;  // 次に供給する type へカーソルを進める（循環）
065:       return;                                                // 供給したので処理終了
066:     }
067: 
068:     this.setOutputData(0, null);                  // 下流が準備できていないので供給なし
069:   }
070: }
071: 
072: menuMixin(SourceNode);
073: window.SourceNode = SourceNode;
074: 
075: // Overlay: 次に供給される予定の情報を表示（下流の可用状態表示は READY/BUSY に統一）
076: SourceNode.prototype.onDrawForeground = function(ctx){
077:   const next = (this._seq && this._seq.length) ? this._seq[this._cursor] : {type:'A'};
078:   // 下流の可用状態（READY/BUSY）に統一（未接続も BUSY とみなす）
079:   let downstream = 'BUSY';
080:   const out = this.outputs && this.outputs[0];
081:   if(out && out.links){
082:     let ready = false;
083:     for(const lid of out.links){
084:       const link = this.graph.links[lid]; if(!link) continue;
085:       const t = this.graph.getNodeById(link.target_id); if(!t) continue;
086:       if(typeof t._state === 'undefined' || t._state === 'IDLE'){ ready = true; break; }
087:     }
088:     downstream = ready ? 'READY' : 'BUSY';
089:   }
090:   const lines = [
091:     `Next: ID=${workCounter+1} Type=${next?next.type:'A'}`,
092:     `Downstream: ${downstream}`,
093:     `SeqLen: ${this._seq?this._seq.length:0} Cursor: ${this._cursor}`,
094:     `workCounter: ${workCounter}`
095:   ];
096:   drawStateBelow(ctx, this, lines, 8, 6);
097: };
